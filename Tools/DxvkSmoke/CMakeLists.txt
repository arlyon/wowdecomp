cmake_minimum_required(VERSION 3.16)
project(DxvkSmoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../../cmake")
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.22)
find_package(DXVKNative REQUIRED)
add_executable(DxvkSmoke Main.cpp)
target_link_libraries(DxvkSmoke PRIVATE PkgConfig::SDL2 DXVKNative::D3D9)
enable_testing()
add_test(NAME DxvkSmoke COMMAND DxvkSmoke)
