cmake_minimum_required(VERSION 3.13)
project(wowdecomp_platform_tests LANGUAGES CXX)
get_filename_component(WOWDECOMP_ROOT "../.." ABSOLUTE)
include("${WOWDECOMP_ROOT}/cmake/WowPlatform.cmake")
wowdecomp_configure_platform()
enable_testing()
add_executable(PlatformTest PlatformTest.cpp)
target_compile_features(PlatformTest PRIVATE cxx_std_17)
target_include_directories(PlatformTest PRIVATE "${WOWDECOMP_ROOT}")
string(TOUPPER "${WOWDECOMP_CPU}" _wow_cpu)
string(TOUPPER "${WOWDECOMP_OS}" _wow_os)
if (_wow_os STREQUAL "DARWIN")
    set(_wow_os MACOS)
endif ()
string(TOUPPER "${WOWDECOMP_COMPILER}" _wow_compiler)
if (_wow_compiler STREQUAL "APPLECLANG")
    set(_wow_compiler CLANG)
elseif (_wow_compiler STREQUAL "GNU")
    set(_wow_compiler GCC)
endif ()
target_compile_definitions(PlatformTest PRIVATE
    EXPECTED_CPU=WOW_CPU_${_wow_cpu} EXPECTED_OS=WOW_PLATFORM_${_wow_os}
    EXPECTED_COMPILER=WOW_COMPILER_${_wow_compiler}
    EXPECTED_POINTER_BITS=${WOWDECOMP_POINTER_BITS})
add_test(NAME PlatformIdentity COMMAND PlatformTest)
foreach (case linux-x64 linux-arm64 macos-arm64 macos-x64 windows-x86 windows-x64 windows-arm64 unknown-os unknown-cpu bad-width universal)
    add_test(NAME PlatformConfig-${case} COMMAND "${CMAKE_COMMAND}"
        "-DCASE=${case}" -P "${CMAKE_CURRENT_SOURCE_DIR}/PlatformConfigTest.cmake")
    if (case MATCHES "^(unknown-|bad-width|universal)")
        set_tests_properties(PlatformConfig-${case} PROPERTIES WILL_FAIL TRUE)
    endif ()
endforeach ()
