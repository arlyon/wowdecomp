cmake_minimum_required(VERSION 3.16)
project(wowdecomp_dxvk_identity_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(WOWDECOMP_ROOT "../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${WOWDECOMP_ROOT}/cmake")
find_package(DXVKNative REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.22)
enable_testing()
add_executable(AdapterIdentityTest AdapterIdentityTest.cpp "${WOWDECOMP_ROOT}/Engine/Source/Gx/DeviceSelection.cpp")
target_include_directories(AdapterIdentityTest PRIVATE "${WOWDECOMP_ROOT}")
target_link_libraries(AdapterIdentityTest PRIVATE DXVKNative::D3D9)
target_compile_definitions(AdapterIdentityTest PRIVATE WOWDECOMP_DXVK=1)
target_compile_options(AdapterIdentityTest PRIVATE -UNDEBUG)
add_test(NAME DxvkAdapterIdentity COMMAND AdapterIdentityTest)
add_executable(AdapterIdentityGpuTest AdapterIdentityGpuTest.cpp)
target_include_directories(AdapterIdentityGpuTest PRIVATE "${WOWDECOMP_ROOT}")
target_link_libraries(AdapterIdentityGpuTest PRIVATE DXVKNative::D3D9 PkgConfig::SDL2)
target_compile_options(AdapterIdentityGpuTest PRIVATE -UNDEBUG)
add_test(NAME DxvkAdapterIdentityGpu COMMAND AdapterIdentityGpuTest)
set_tests_properties(DxvkAdapterIdentityGpu PROPERTIES LABELS "GPU" TIMEOUT 60
    ENVIRONMENT "DXVK_WSI_DRIVER=SDL2")
